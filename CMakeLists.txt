cmake_minimum_required(VERSION 3.16)
project(WorldSimulation CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(WSIM_SANITIZE "Build with AddressSanitizer and UBSan" OFF)
if(WSIM_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra)

add_executable(wsim src/main.cpp)
target_include_directories(wsim PRIVATE include)
find_package(SFML 3 COMPONENTS Graphics QUIET)
if(SFML_FOUND)
  target_link_libraries(wsim PRIVATE SFML::Graphics)
  message(STATUS "SFML 3 found: building the viewer")
else()
  find_package(SFML 2.5 COMPONENTS graphics window system QUIET)
  if(SFML_FOUND)
    target_link_libraries(wsim PRIVATE sfml-graphics sfml-window sfml-system)
    message(STATUS "SFML 2 found: building the viewer")
  else()
    target_compile_definitions(wsim PRIVATE WSIM_NO_SFML)
    message(STATUS "SFML not found: headless build only")
  endif()
endif()

enable_testing()
find_path(DOCTEST_INCLUDE doctest/doctest.h)
if(DOCTEST_INCLUDE)
  add_executable(wsim_tests tests/test_wsim.cpp)
  target_include_directories(wsim_tests PRIVATE include ${DOCTEST_INCLUDE})
  target_compile_definitions(wsim_tests PRIVATE WSIM_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME wsim_tests COMMAND wsim_tests)
endif()
